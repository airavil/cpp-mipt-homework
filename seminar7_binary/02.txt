1) int a = 0x11223344;
   44 33 22 11

2) int b = 1000000000;
   00 CA 9A 3B

3) int c = 65535;
   FF FF 00 00

4) int d = -1;
   FF FF FF FF

5) int array[3] = {10, 2000, 65535};
   10    = 0x0000000A -> 0A 00 00 00
   2000  = 0x000007D0 -> D0 07 00 00
   65535 = 0x0000FFFF -> FF FF 00 00
   Итого: 0A 00 00 00 D0 07 00 00 FF FF 00 00

6) char str[9] = "Cat\nDog\n";
   C = 0x43, a = 0x61, t = 0x74, \n = 0x0A,
   D = 0x44, o = 0x6F, g = 0x67, \n = 0x0A, '\0' = 0x00
   43 61 74 0A 44 6F 67 0A 00

7) float x = 1.0f;
   00 00 80 3F
